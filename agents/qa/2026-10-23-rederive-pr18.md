# QA re-derivation, PR #18 (2026-10-23)

This is an independent check of `dividend-recap-irr`, `fund-waterfall-carry` and `cet1-ratio`, with seeds `qa-0` to `qa-1999` for each template (6,000 instances). The script reads the numbers from the rendered statement and recomputes each answer without using `src/lib/finance`.

Result: 0 mismatches. The waterfall instances fell in all three zones: full 1,553, below hurdle 302, catch-up 145.

Run from the repo root: `npx tsx --tsconfig tsconfig.json <this script saved as .ts>`.

```ts
import { instantiate } from "./src/lib/problems/index";
const txt = (inst: any) => inst.blocks.map((b: any) => b.text ?? JSON.stringify(b.rows ?? "")).join(" ");
const nums = (s: string) => [...s.matchAll(/\*\*\$?([\d,]+(?:\.\d+)?)m?\*\*/g)].map((m) => +m[1].replace(/,/g, ""));
function irr(f: number[]) { let lo = -0.99, hi = 5; const npv = (r: number) => f.reduce((s, x, t) => s + x / (1 + r) ** t, 0);
  for (let i = 0; i < 300; i++) { const m = (lo + hi) / 2; if (npv(lo) * npv(m) <= 0) hi = m; else lo = m; } return (lo + hi) / 2; }
let bad = 0, n = 0; const zones: Record<string, number> = {};
for (let i = 0; i < 2000; i++) {
  const seed = `qa-${i}`;
  // dividend recap
  let a = instantiate("dividend-recap-irr", seed); let s = txt(a);
  const yr = +s.match(/end of \*\*year (\d)\*\*/)![1]; const [e0, d, x] = nums(s);
  const f = [-e0, 0, 0, 0, 0, x]; f[yr] = d; const mine = irr(f) * 100;
  n++; if (Math.abs(mine - a.answer) > 0.005 || !isFinite(a.answer)) { bad++; if (bad < 5) console.log("recap", seed, mine, a.answer, s); }
  // waterfall
  a = instantiate("fund-waterfall-carry", seed); s = txt(a);
  const c = nums(s)[0], T = +s.match(/After \*\*(\d+) years/)![1], D = +s.match(/distributes \*\*\$?([\d,.]+)m/)![1].replace(/,/g, "");
  const pref = c * (1.08 ** T - 1); let left = D - c; let gp = 0;
  if (left > pref) { left -= pref; const cu = Math.min(left, pref / 4); gp = cu + 0.2 * (left - cu); }
  // sanity: when complete, gp = 0.2*(D-c)
  const z = D - c <= pref ? "below" : D - c < pref * 1.25 ? "catchup" : "full"; zones[z] = (zones[z] ?? 0) + 1;
  n++; if (Math.abs(gp - a.answer) > 0.005 || (z === "full" && Math.abs(gp - 0.2 * (D - c)) > 1e-6)) { bad++; if (bad < 5) console.log("wf", seed, gp, a.answer, s); }
  // cet1
  a = instantiate("cet1-ratio", seed); const rows = (a.blocks[1] as any).rows as string[][];
  const w = [0, 0.35, 0.75, 1]; const credit = rows.reduce((t, r, k) => t + +r[1].replace(/,/g, "") * w[k], 0);
  const rw = rows.map((r) => r[2]).join(","); if (rw !== "0%,35%,75%,100%") { bad++; console.log("weights", rw); }
  s = txt(a); const other = +s.match(/add \*\*\$?([\d,.]+)m/)![1].replace(/,/g, ""), cet = +s.match(/holds \*\*\$?([\d,.]+)m/)![1].replace(/,/g, "");
  const r = cet / (credit + other) * 100;
  n++; if (Math.abs(r - a.answer) > 0.005) { bad++; if (bad < 5) console.log("cet1", seed, r, a.answer, s); }
}
console.log({ checked: n, mismatches: bad, waterfallZones: zones });
```
